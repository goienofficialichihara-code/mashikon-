  <script>
    const CONFIG = { totalMale: 10, totalFemale: 10 };

    let state = {
      currentUser: null,
      votes: {},
      isCalculated: false,
      matchingResults: {}
    };

    window.addEventListener('DOMContentLoaded', () => {
      loadState();
      setupDropdowns();
      updateAdminMonitor();

      window.addEventListener('storage', (e) => {
        if (e.key === 'machikon_app_state') {
          loadState();
          renderScreen();
          updateAdminMonitor();
        }
      });
    });

    function saveState() {
      localStorage.setItem('machikon_app_state', JSON.stringify(state));
      renderScreen();
      updateAdminMonitor();
    }

    function loadState() {
      const saved = localStorage.getItem('machikon_app_state');
      if (saved) {
        state = JSON.parse(saved);
      }
    }

    function setupDropdowns() {
      const numSelect = document.getElementById('user-number');
      numSelect.innerHTML = '';
      for (let i = 1; i <= CONFIG.totalMale; i++) {
        numSelect.innerHTML += `<option value="${i}">${i} 番</option>`;
      }
    }

    function populateVoteOptions() {
      if (!state.currentUser) return;
      const targetGender = state.currentUser.gender === 'male' ? 'female' : 'male';
      const targetPrefix = targetGender === 'male' ? '男子' : '女子';
      const total = targetGender === 'male' ? CONFIG.totalMale : CONFIG.totalFemale;

      ['vote-1', 'vote-2', 'vote-3'].forEach(id => {
        const select = document.getElementById(id);
        const currentVal = select.value;
        select.innerHTML = '<option value="">-- 選択なし --</option>';
        for (let i = 1; i <= total; i++) {
          const val = `${targetGender}-${i}`;
          select.innerHTML += `<option value="${val}">${targetPrefix} ${i} 番</option>`;
        }
        select.value = currentVal;
      });
    }

    function updateVoteDropdowns() {
      const v1 = document.getElementById('vote-1').value;
      const v2 = document.getElementById('vote-2').value;
      const v3 = document.getElementById('vote-3').value;

      const selected = [v1, v2, v3].filter(Boolean);

      ['vote-1', 'vote-2', 'vote-3'].forEach((id) => {
        const select = document.getElementById(id);
        const thisVal = select.value;
        Array.from(select.options).forEach(opt => {
          if (opt.value && opt.value !== thisVal && selected.includes(opt.value)) {
            opt.disabled = true;
          } else {
            opt.disabled = false;
          }
        });
      });
    }

    function handleLogin(e) {
      e.preventDefault();
      const form = e.target;
      const gender = form.gender.value;
      const num = parseInt(document.getElementById('user-number').value);
      
      state.currentUser = { gender, num, id: `${gender}-${num}` };
      saveState();
    }

    function logout() {
      state.currentUser = null;
      saveState();
    }

    function renderScreen() {
      const secLogin = document.getElementById('sec-login');
      const secVote = document.getElementById('sec-vote');
      const secResult = document.getElementById('sec-result');

      if (!state.currentUser) {
        secLogin.classList.remove('hidden');
        secVote.classList.add('hidden');
        secResult.classList.add('hidden');
        return;
      }

      secLogin.classList.add('hidden');
      const userId = state.currentUser.id;

      if (state.isCalculated) {
        secVote.classList.add('hidden');
        secResult.classList.remove('hidden');

        document.getElementById('state-waiting').classList.add('hidden');
        document.getElementById('state-published').classList.remove('hidden');

        const partnerId = state.matchingResults[userId];
        if (partnerId) {
          const [pGender, pNum] = partnerId.split('-');
          const pLabel = (pGender === 'male' ? '男子 ' : '女子 ') + pNum + ' 番';
          document.getElementById('matched-partner-number').innerText = pLabel;
          document.getElementById('result-matched').classList.remove('hidden');
          document.getElementById('result-unmatched').classList.add('hidden');
        } else {
          document.getElementById('result-matched').classList.add('hidden');
          document.getElementById('result-unmatched').classList.remove('hidden');
        }
        return;
      }

      if (state.votes[userId]) {
        secVote.classList.add('hidden');
        secResult.classList.remove('hidden');
        document.getElementById('state-waiting').classList.remove('hidden');
        document.getElementById('state-published').classList.add('hidden');
      } else {
        secVote.classList.remove('hidden');
        secResult.classList.add('hidden');
        
        const gLabel = state.currentUser.gender === 'male' ? '男子' : '女子';
        const badge = document.getElementById('user-badge');
        badge.innerText = `${gLabel} ${state.currentUser.num} 番でログイン中`;
        badge.className = `px-3 py-1 rounded-full text-xs font-bold ${state.currentUser.gender === 'male' ? 'bg-blue-100 text-blue-700' : 'bg-pink-100 text-pink-700'}`;

        populateVoteOptions();
      }
    }

    function handleVoteSubmit(e) {
      e.preventDefault();
      if (!state.currentUser) return;

      const userId = state.currentUser.id;
      const choice1 = document.getElementById('vote-1').value;
      const choice2 = document.getElementById('vote-2').value;
      const choice3 = document.getElementById('vote-3').value;

      state.votes[userId] = { choice1, choice2, choice3 };
      saveState();
    }

    function updateAdminMonitor() {
      let maleVotedCount = 0;
      let femaleVotedCount = 0;

      const maleGrid = document.getElementById('grid-male-status');
      const femaleGrid = document.getElementById('grid-female-status');
      maleGrid.innerHTML = '';
      femaleGrid.innerHTML = '';

      for (let i = 1; i <= CONFIG.totalMale; i++) {
        const id = `male-${i}`;
        const isVoted = !!state.votes[id];
        if (isVoted) maleVotedCount++;
        maleGrid.innerHTML += `
          <div class="p-1 text-center rounded ${isVoted ? 'bg-blue-600 text-white font-bold' : 'bg-slate-200 text-slate-400'}">
            ${i}
          </div>
        `;
      }

      for (let i = 1; i <= CONFIG.totalFemale; i++) {
        const id = `female-${i}`;
        const isVoted = !!state.votes[id];
        if (isVoted) femaleVotedCount++;
        femaleGrid.innerHTML += `
          <div class="p-1 text-center rounded ${isVoted ? 'bg-pink-600 text-white font-bold' : 'bg-slate-200 text-slate-400'}">
            ${i}
          </div>
        `;
      }

      document.getElementById('count-male-voted').innerText = `${maleVotedCount} / ${CONFIG.totalMale}`;
      document.getElementById('count-female-voted').innerText = `${femaleVotedCount} / ${CONFIG.totalFemale}`;

      const couplesBox = document.getElementById('admin-results-box');
      const couplesList = document.getElementById('admin-couples-list');
      if (state.isCalculated) {
        couplesBox.classList.remove('hidden');
        couplesList.innerHTML = '';
        
        const displayed = new Set();
        let count = 0;
        Object.entries(state.matchingResults).forEach(([uId, pId]) => {
          if (!displayed.has(uId) && !displayed.has(pId)) {
            displayed.add(uId);
            displayed.add(pId);
            count++;
            
            const [g1, n1] = uId.split('-');
            const [g2, n2] = pId.split('-');
            const mNum = g1 === 'male' ? n1 : n2;
            const fNum = g1 === 'female' ? n1 : n2;

            couplesList.innerHTML += `
              <div class="p-2.5 bg-emerald-50 rounded-lg border border-emerald-200 flex justify-between items-center font-bold text-slate-700">
                <span>ペア ${count}</span>
                <span class="text-emerald-700">男子 ${mNum} 番  <i class="fa-solid fa-heart text-rose-500 mx-1"></i>  女子 ${fNum} 番</span>
              </div>
            `;
          }
        });
        if (count === 0) {
          couplesList.innerHTML = `<p class="text-slate-400 italic">成立したカップルはありませんでした。</p>`;
        }
      } else {
        couplesBox.classList.add('hidden');
      }
    }

    function generateRandomVotes() {
      for (let i = 1; i <= CONFIG.totalMale; i++) {
        const id = `male-${i}`;
        state.votes[id] = getRandomChoice('female', CONFIG.totalFemale);
      }
      for (let i = 1; i <= CONFIG.totalFemale; i++) {
        const id = `female-${i}`;
        state.votes[id] = getRandomChoice('male', CONFIG.totalMale);
      }
      saveState();
    }

    function getRandomChoice(targetGender, total) {
      const nums = Array.from({length: total}, (_, i) => i + 1).sort(() => Math.random() - 0.5);
      return {
        choice1: `${targetGender}-${nums[0]}`,
        choice2: `${targetGender}-${nums[1]}`,
        choice3: `${targetGender}-${nums[2]}`
      };
    }

    function runMatchingAlgorithm() {
      const candidates = [];

      Object.entries(state.votes).forEach(([userId, v]) => {
        if (!userId.startsWith('male-')) return;
        const maleId = userId;

        [v.choice1, v.choice2, v.choice3].forEach((femaleId, idx) => {
          if (!femaleId) return;
          const mRank = idx + 1;

          const fVote = state.votes[femaleId];
          if (!fVote) return;

          let fRank = 0;
          if (fVote.choice1 === maleId) fRank = 1;
          else if (fVote.choice2 === maleId) fRank = 2;
          else if (fVote.choice3 === maleId) fRank = 3;

          if (fRank > 0) {
            candidates.push({
              maleId,
              femaleId,
              score: mRank + fRank,
              mRank,
              fRank
            });
          }
        });
      });

      candidates.sort((a, b) => {
        if (a.score !== b.score) return a.score - b.score;

        const aHasRank1 = (a.mRank === 1 || a.fRank === 1);
        const bHasRank1 = (b.mRank === 1 || b.fRank === 1);
        if (aHasRank1 && !bHasRank1) return -1;
        if (!aHasRank1 && bHasRank1) return 1;

        return Math.random() - 0.5;
      });

      const matchedResults = {};
      const matchedMales = new Set();
      const matchedFemales = new Set();

      candidates.forEach(c => {
        if (!matchedMales.has(c.maleId) && !matchedFemales.has(c.femaleId)) {
          matchedMales.add(c.maleId);
          matchedFemales.add(c.femaleId);
          matchedResults[c.maleId] = c.femaleId;
          matchedResults[c.femaleId] = c.maleId;
        }
      });

      state.matchingResults = matchedResults;
      state.isCalculated = true;
      saveState();
      alert('集計が完了しました！参加者の画面に結果が一斉送信されました。');
    }

    function resetAllData() {
      if (confirm('すべての投票データと集計結果をリセットしますか？')) {
        state = {
          currentUser: null,
          votes: {},
          isCalculated: false,
          matchingResults: {}
        };
        saveState();
      }
    }

    function toggleAdminModal() {
      const modal = document.getElementById('modal-admin');
      if (modal.classList.contains('hidden')) {
        const inputPass = prompt('主催者パスワードを入力してください:');
        if (inputPass === '0323') {
          modal.classList.remove('hidden');
        } else if (inputPass !== null) {
          alert('パスワードが違います。');
        }
      } else {
        modal.classList.add('hidden');
      }
    }
